Installation

Get started with the Reusable Components Library in your React project

1

Install the Package

npm install reusable-components
# or
yarn add reusable-components
2

Import Styles

/* In your main CSS file */
@import 'tailwindcss/base';
@import 'tailwindcss/components';
@import 'tailwindcss/utilities';
3

Configure Tailwind (if not already done)

// tailwind.config.js
module.exports = {
  content: [
    "./src/**/*.{js,jsx,ts,tsx}",
    "./node_modules/reusable-components/**/*.{js,jsx,ts,tsx}"
  ],
  theme: {
    extend: {},
  },
  plugins: [],
}
4

Start Using Components

import React from 'react';
import { Button, Input, Modal } from 'reusable-components';

function App() {
  return (
    <div>
      <Button variant="primary">
        Click me!
      </Button>
    </div>
  );
}

Quick Start

Build your first form with our components in minutes

Live Preview

import React, { useState } from 'react';
import { Input, Select, Button, FormLayout } from 'reusable-components';

function UserForm() {
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    role: ''
  });

  return (
    <FormLayout title="User Registration">
      <Input
        label="Full Name"
        placeholder="Enter your name"
        value={formData.name}
        onChange={(e) => setFormData({...formData, name: e.target.value})}
        required
      />
      
      <Input
        type="email"
        label="Email"
        placeholder="you@example.com"
        value={formData.email}
        onChange={(e) => setFormData({...formData, email: e.target.value})}
        required
      />
      
      <Select
        label="Role"
        placeholder="Select role..."
        options={[
          { value: 'admin', label: 'Admin' },
          { value: 'user', label: 'User' }
        ]}
        value={formData.role}
        onChange={(value) => setFormData({...formData, role: value})}
      />
      
      <Button variant="primary" type="submit">
        Submit
      </Button>
    </FormLayout>
  );
}

import React, { useState } from 'react';
import { 
  SearchBox, 
  FilterPanel, 
  EditableDataGrid,
  BulkActions 
} from 'reusable-components';

function UserManagement() {
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedRows, setSelectedRows] = useState([]);

  const columns = [
    { key: 'name', title: 'Name', sortable: true },
    { key: 'email', title: 'Email', sortable: true },
    { key: 'role', title: 'Role', filterable: true },
    { key: 'status', title: 'Status', editable: true }
  ];

  const data = [
    { 
      id: '1', 
      name: 'John Doe', 
      email: 'john@example.com', 
      role: 'Admin', 
      status: 'Active' 
    },
    // ... more data
  ];

  return (
    <div>
      <SearchBox
        placeholder="Search users..."
        onSearch={setSearchTerm}
      />
      
      <FilterPanel
        filters={[
          {
            key: 'role',
            label: 'Role',
            type: 'select',
            options: [
              { value: 'admin', label: 'Admin' },
              { value: 'user', label: 'User' }
            ]
          }
        ]}
      />
      
      <BulkActions
        selectedItems={selectedRows}
        actions={[
          { key: 'activate', label: 'Activate' },
          { key: 'deactivate', label: 'Deactivate' },
          { key: 'delete', label: 'Delete', variant: 'danger' }
        ]}
      />
      
      <EditableDataGrid
        columns={columns}
        data={data}
        selection={{
          type: 'checkbox',
          selectedRowKeys: selectedRows,
          onSelectionChange: setSelectedRows
        }}
        sortable
        filterable
        editable
      />
    </div>
  );
}
